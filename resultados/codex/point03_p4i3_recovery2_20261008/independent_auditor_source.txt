def audit(out):
    old = retained_audit(ROOT / 'resultados/codex/point03_p4i2_gaussian_time_20261008')
    assert old['integrity_pass'] and (not old['temporal_precision_pass'])
    e = read(out / 'execution.json')
    m = read(out / 'manifest.json')
    a = read(out / 'assessment.json')
    assert e['status'] == 'completed' and e['sources_unchanged'] and (e['elapsed_s'] <= 22000)
    assert len(e['children']) == len(e['checkpoints']) == 256
    assert m['plan'] == [dict(id='R5_65536', method='R5', steps=65536)] and m['reference_label'] == 'P6_32768'
    assert sha(out / 'manifest.json') == e['manifest_sha256'] and sha(out / 'assessment.json') == e['assessment_sha256']
    for p, v in m['source_sha256'].items():
        assert sha(p) == v
    det = Path(m['detector_folder'])
    with np.load(det / 'gaussian_q16.npz', allow_pickle=False) as data:
        free = data['free_dofs'].copy()
    M = load_npz(Path(m['geometry_folder']) / 'mass.npz')[free, :][:, free]
    C = load_npz(det / 'core_q32.npz')[free, :][:, free]
    w = np.load(det / 'intensity_q32.npy', allow_pickle=False)[free]
    previous = digest = None
    power = m['P_in']
    residual = 0.0
    for index, link in enumerate(e['checkpoints'], 1):
        cp = read(link['path'])
        event = e['children'][index - 1]
        assert sha(link['path']) == link['sha256'] and cp['status'] == 'completed' and (cp['case'] == 'R5_65536') and (cp['index'] == index)
        assert event['index'] == index and event['returncode'] == 0 and (event['elapsed_s'] <= 370) and (cp['elapsed_s'] <= 360)
        assert cp['previous_report_path'] == previous and cp['previous_report_sha256'] == digest
        assert cp['steps_done'] == 256 * index and cp['steps_total'] == 65536 and (cp['dt_m'] == 0.002 / 65536)
        assert cp['started_utc'] > (m['original_manifest_created_utc'] if index <= 74 else m['first_recovery_manifest_created_utc'] if index <= 89 else m['created_utc'])
        for sample in ('resource_start', 'resource_end'):
            assert cp[sample]['available_ram_bytes'] > 3 * 1024 ** 3 and cp[sample]['free_disk_bytes'] > 2 * 1024 ** 3
        assert sha(cp['field_path']) == cp['field_sha256']
        with np.load(cp['field_path'], allow_pickle=False) as data:
            field = data['field'].copy()
        assert field.shape == (len(free),) and field.dtype == np.dtype('complex128') and np.all(np.isfinite(field))
        actual = float(np.vdot(field, M @ field).real)
        residual = max(residual, abs(actual - cp['physical_power']))
        assert abs(actual - cp['physical_power']) <= 1e-12 and abs(power - cp['previous_power']) <= 1e-12 and (0 < actual <= power + 1e-09)
        previous, digest, power = (link['path'], link['sha256'], actual)
    fields = {'R5': field}
    for label, links in (('P6', m['reference']['checkpoints']), ('R5_old', m['previous_Radau_reference']['checkpoints'])):
        cp = read(links[-1]['path'])
        assert sha(cp['field_path']) == cp['field_sha256']
        with np.load(cp['field_path'], allow_pickle=False) as data:
            fields[label] = data['field'].copy()
    powers = {label: dict(field=float(np.vdot(f, C @ f).real), intensity=float(w @ abs(f) ** 2)) for label, f in fields.items()}
    d = fields['P6'] - fields['R5']
    relative = math.sqrt(float(np.vdot(d, M @ d).real) / float(np.vdot(fields['R5'], M @ fields['R5']).real))
    bounds = {}
    changes = {}
    for method in ('field', 'intensity'):
        squared = float(np.vdot(d, C @ d).real) if method == 'field' else float(w @ abs(d) ** 2)
        assert squared >= -1e-20
        bounds[method] = (math.sqrt(powers['P6'][method]) + math.sqrt(powers['R5'][method])) * math.sqrt(max(0.0, squared))
        changes[method] = abs(powers['P6'][method] - powers['R5'][method])
        assert abs(bounds[method] - a['PSD_observable_bounds'][method]) <= 1e-12 and abs(changes[method] - a['power_differences'][method]) <= 1e-12
        for label in powers:
            assert abs(powers[label][method] - a['powers'][label][method]) <= 1e-12
    diagnostic = fields['R5_old'] - fields['R5']
    rd = math.sqrt(float(np.vdot(diagnostic, M @ diagnostic).real) / float(np.vdot(fields['R5'], M @ fields['R5']).real))
    assert abs(relative - a['relative_field_difference']) <= 1e-12 and abs(rd - a['diagnostic_Radau_refinement_relative_field']) <= 1e-12
    passed = relative <= 0.0001 and max(bounds.values()) <= 1e-05 and (max(changes.values()) <= 1e-06)
    assert passed == a['temporal_precision_pass'] == e['temporal_precision_pass']
    return dict(created_utc=datetime.now(timezone.utc).isoformat(), integrity_pass=True, temporal_precision_pass=passed, point03_closed=False, new_checkpoints_checked=167, retained_checkpoints_rechecked=441, max_power_residual=residual, powers=powers, relative_field_difference=relative, PSD_observable_bounds=bounds, power_differences=changes, diagnostic_Radau_refinement_relative_field=rd, previous_P4I_pass=False, previous_P4I2_pass=False, scope='One-mesh longitudinal precision only; spatial and domain closure still required.')
